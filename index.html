CÓDIGO ARDUINO / ESP32 (BOTÕES NF)

Grave este código no ESP32 usando a IDE Arduino para botões Normalmente Fechados (NF) nos pinos GPIO 4 e 5.


// Código C++ Otimizado para ESP32 (Botões NF com Debounce)
const int PIN_P1 = 4;
const int PIN_P2 = 5;

int lastStateP1 = LOW;
int lastStateP2 = LOW;
unsigned long lastDebounceTimeP1 = 0;
unsigned long lastDebounceTimeP2 = 0;
const unsigned long debounceDelay = 50;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_P1, INPUT_PULLUP);
  pinMode(PIN_P2, INPUT_PULLUP);
}

void loop() {
  int readingP1 = digitalRead(PIN_P1);
  int readingP2 = digitalRead(PIN_P2);

  if (readingP1 != lastStateP1) {
    lastDebounceTimeP1 = millis();
  }
  if ((millis() - lastDebounceTimeP1) > debounceDelay) {
    if (readingP1 == HIGH && lastStateP1 == LOW) {
      Serial.println("P1");
      delay(300);
    }
  }
  lastStateP1 = readingP1;

  if (readingP2 != lastStateP2) {
    lastDebounceTimeP2 = millis();
  }
  if ((millis() - lastDebounceTimeP2) > debounceDelay) {
    if (readingP2 == HIGH && lastStateP2 == LOW) {
      Serial.println("P2");
      delay(300);
    }
  }
  lastStateP2 = readingP2;
}
